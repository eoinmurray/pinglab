export const nativeColumns = ["PING"];
export const nativeEvidence = await reportSnapshots('exp046', { snapshots: {"PING": "infer/ping__tg4p5__seed42/inference/start/snapshot.json"}, allowUnavailable: true });
export const nativeValue = value => {
  if (value === undefined || value === null) return 'Unavailable';
  if (Array.isArray(value)) return value.map(nativeValue).join(' / ');
  if (typeof value === 'object') return value.value === undefined ? JSON.stringify(value) : `${nativeValue(value.value)} ${value.unit ?? ''}`.trim();
  return String(value);
};
export const nativePopulation = (name, id) => nativeEvidence?.snapshots[name]?.network?.populations?.find(row => row.id === id);
export const nativeProjection = (name, id) => nativeEvidence?.snapshots[name]?.network?.projections?.find(row => row.id === id);
export const nativeRow = (label, select) => [label, ...nativeColumns.map(name => nativeValue(select(name)))];
export const nativePair = (name, key) => ['E', 'I'].map(id => nativeEvidence?.snapshots[name] && !nativePopulation(name, id) ? 'Not applicable' : nativePopulation(name, id)?.neuron?.[key]);
export const nativeConfig = nativeEvidence?.configuration;
export const recordedSetting = (...paths) => {
  for (const path of paths) {
    const value = path.split('.').reduce((object, key) => object?.[key], nativeConfig);
    if (value !== undefined) return value;
  }
  return undefined;
};

## Parameters

Model values are read from original native snapshots through the selected
presentation's compute ancestry. Paired values are E / I. Initializer parent means and SDs describe the original
weight distributions, in µS for conductance projections and dimensionless for
the current-based readout; they are not measured trained-weight statistics. Missing snapshots
or fields are unavailable; current recipes and historical copied tables are
not substituted. The recorded configuration identifies the input, numerical
and analysis choices of that computation; it does not establish a separate
upstream training protocol.

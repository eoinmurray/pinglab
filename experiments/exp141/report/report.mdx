import RunFigure from '../../../site/src/components/RunFigure.astro';
import { reportData, percent, round } from '../../../site/src/lib/report-data';

export const data = await reportData('exp141', { allowUnavailable: true });
export const trainingSummary = mode => {
  if (!data) return 'Local presentation data is unavailable.';
  const rows = data.curves[mode];
  const first = rows[0];
  const last = rows.at(-1);
  const best = Math.max(...rows.map(row => row.validation.accuracy));
  const training = last.train;
  const test = data.test[mode];
  return `After ${last.epoch} epochs, online training accuracy was ${percent(training.accuracy)} and validation accuracy was ${percent(last.validation.accuracy)}. Best validation accuracy was ${percent(best)}. Validation cross-entropy changed from ${round(first.validation.loss, 3)} to ${round(last.validation.loss, 3)}. The checkpoint selected by minimum validation loss was epoch ${test.epoch}; its held-out test accuracy was ${percent(test.accuracy)} on ${test.examples} images. The black curve shows online pre-update minibatch metrics; the red curve uses fixed epoch-end weights. Epoch zero shows untrained validation only. Crosses mark the selected-checkpoint test measurements.`;
};
export const rasterSummary = mode => {
  if (!data) return 'The response will be recorded after training from the validation-selected checkpoint.';
  const sample = data.sample;
  const result = sample.modes[mode];
  return `Original sample ${sample.index} is digit ${sample.label}; the validation-selected model (epoch ${result.epoch}) predicted ${result.prediction}. This image belongs to the ${sample.partition} partition. It is a single-example diagnostic, not an independent test result.`;
};

## Abstract

We compare two spiking networks on small handwritten digits. Both use
conductance-based leaky integrate-and-fire neurons and choose a digit by counting
spikes from ten output neurons. The PING model adds a fixed inhibitory feedback
loop to the feed-forward COBA model. Both use the same training, validation and test partitions,
one random seed and classification loss without spike penalties. With 200 ms
presentations and 30 epochs, validation-selected COBA and PING checkpoints
reached {data ? percent(data.test.coba.accuracy) : 'unavailable'} and
{data ? percent(data.test.ping.accuracy) : 'unavailable'} held-out test accuracy,
respectively. Adding feedback did not improve classification in this run;
variation across training seeds was not tested.

## Results

Recipe v9 ran locally on CPU with SNNLab 0.23.0 and autocast disabled.
The dataset contained {data ? data.dataset.train_samples : 'unavailable'} training,
{data ? data.dataset.validation_samples : 'unavailable'} validation and
{data ? data.dataset.test_samples : 'unavailable'} held-out test images.
Each model trained for 30 epochs; minimum validation cross-entropy selected
one checkpoint for a single held-out test evaluation. The rasters replay one
training image through those selected checkpoints.

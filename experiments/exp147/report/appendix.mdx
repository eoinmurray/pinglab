import Table from '../../../site/src/components/Table.astro';
import RunFigure from '../../../site/src/components/RunFigure.astro';
import { reportSnapshots } from '../../../site/src/lib/report-data';

export const evidence = await reportSnapshots('exp147', {
  snapshots: { simulation: 'simulation/start/snapshot.json' },
  allowUnavailable: true,
});
export const cfg = evidence?.configuration;
export const snapshot = evidence?.snapshots.simulation;
export const graph = snapshot?.network;
export const neuron = graph?.populations?.find(row => row.id === 'n1')?.neuron?.config;
export const synapse = graph?.projections?.find(row => row.id === 'n1_to_n2')?.synapse?.config;
export const show = value => value === undefined || value === null ? 'Unavailable' : String(value);
export const pair = (a, b) => `${show(a)} / ${show(b)}`;
export const groups = [
  { label: 'Network', rows: [
    ['Neurons / reciprocal synapses', pair(graph?.populations?.length, graph?.projections?.length)],
    ['HCOs executed together', show(cfg?.batch_size)],

  ] },
  { label: 'Neurons', rows: [
    ['Capacitance (nF)', show(neuron?.capacitance_nf)],
    ['Leak conductance (µS) / reversal (mV)', pair(neuron?.leak_us, neuron?.leak_reversal_mv)],
    ['Na / K maximum conductance (µS)', pair(neuron?.g_na_us, neuron?.g_k_us)],
    ['Na / K reversal (mV)', pair(neuron?.e_na_mv, neuron?.e_k_mv)],
    ['Na / K gate time constant (ms)', pair(neuron?.tau_na_ms, neuron?.tau_k_ms)],
    ['Na / K sigmoid slope (mV⁻¹)', pair(neuron?.gamma_na_per_mv, neuron?.gamma_k_per_mv)],
    ['Na / K sigmoid midpoint (mV)', pair(neuron?.bias_na_mv, neuron?.bias_k_mv)],
    ['Initial membrane and ionic gate voltages (mV)', show(neuron?.initial_voltage_mv)],
    ['Observed upward-crossing threshold (mV)', show(neuron?.spike_threshold_mv)],
  ] },
  { label: 'Synapses', rows: [
    ['Maximum conductance (µS)', show(synapse?.g_max_us)],
    ['Inhibitory reversal potential (mV)', show(neuron?.synapse_reversal_mv)],
    ['Gate time constant (ms)', show(synapse?.tau_ms)],
    ['Sigmoid slope (mV⁻¹) / midpoint (mV)', pair(synapse?.gamma_per_mv, synapse?.bias_mv)],

  { label: 'Input and simulation', rows: [

];

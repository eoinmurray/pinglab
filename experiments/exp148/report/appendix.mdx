import Table from '../../../site/src/components/Table.astro';
import RunFigure from '../../../site/src/components/RunFigure.astro';
import { reportSnapshots } from '../../../site/src/lib/report-data';

export const evidence = await reportSnapshots('exp148', {
  analysisInput: 'analyse',
  computeInput: 'compute',
  snapshots: { simulation: 'simulation/start/snapshot.json' },
  allowUnavailable: true,
});
export const parameterGroups = (() => {
  if (!evidence) return [{ label: 'Network', rows: [['Native snapshot evidence', 'Unavailable']] }];
  const cfg = evidence.configuration;
  const snapshot = evidence.snapshots.simulation;
  const graph = snapshot?.network;
  const population = name => graph?.populations?.find(row => row.id === name);
  const projection = name => graph?.projections?.find(row => row.id === name);
  const parameter = name => graph?.parameters?.find(row => row.id === `${name}.weight`);
  const show = value => {
    const scalar = value?.value ?? value;
    if (scalar === undefined || scalar === null) return 'Unavailable';
    return typeof scalar === 'number' ? String(Number(scalar.toPrecision(6))) : String(scalar);
  };
  const pair = key => ['E', 'I'].map(name => show(population(name)?.neuron?.[key])).join(' / ');
  const names = ['E_to_E', 'E_to_I', 'I_to_E', 'I_to_I', 'X_to_E', 'X_to_I'];
  const label = name => name.replace(/^X_to_/, 'Input→').replace('_to_', '→').replace('.weight', ' weight');
  const density = name => {
    const zero = snapshot?.execution?.initialization?.[`${name}.weight`]?.statistics?.zero_fraction;
    return zero === undefined ? 'Unavailable' : show(1 - zero);
  };
  return [
    { label: 'Network', rows: [
      ['E / I neurons', ['E', 'I'].map(name => show(population(name)?.size)).join(' / ')],
      ['Shared Input sources', show(graph?.inputs?.find(row => row.id === 'X')?.shape?.at(-1))],
      ...names.map(name => [`${label(name)} autapses allowed`, show(projection(name)?.allow_autapses)]),
    ] },
    { label: 'Neurons', rows: [
      ['Neuron model, E / I', pair('kind')],
      ['Capacitance, E / I (nF)', pair('capacitance_nf')],
      ['Leak conductance, E / I (µS)', pair('leak_us')],
      ['Derived time constants, E / I (ms)', pair('tau_mem')],
      ['Resting potential, E / I (mV)', pair('resting_mv')],
      ['Initial voltage, E / I (mV)', pair('initial_voltage_mv')],
      ['Reset potential, E / I (mV)', pair('reset_mv')],
      ['Spike threshold, E / I (mV)', pair('threshold_mv')],
      ['Excitatory reversal, E / I (mV)', pair('excitatory_reversal_mv')],
      ['Inhibitory reversal, E / I (mV)', pair('inhibitory_reversal_mv')],
      ['Refractory duration, E / I (ms)', pair('refractory')],
      ['Refractory scheduling', show(graph?.refractory_timing)],
    ] },
    { label: 'Synapses and weights', rows: names.flatMap(name => {
      const spec = parameter(name);
      const init = spec?.initializer;
      const stats = snapshot?.execution?.initialization?.[`${name}.weight`]?.statistics;
      return [
        [`${label(name)} synapse / decay / delay (ms)`, [projection(name)?.synapse?.kind, projection(name)?.synapse?.tau, projection(name)?.delay].map(show).join(' / ')],
        [`${label(name)} modulation`, projection(name)?.modulation?.kind ?? 'None'],
        [`${label(name)} initializer / scaling`, [init?.kind, spec?.initialization_scaling].map(show).join(' / ')],
        [`${label(name)} compensated initializer mean / SD (µS)`, [init?.mean, init?.std].map(show).join(' / ')],
        [`${label(name)} connected Gaussian parent mean / SD (µS)`, init && typeof init.initial_zero_fraction === 'number' ? [init.mean / (1 - init.initial_zero_fraction), init.std / (1 - init.initial_zero_fraction)].map(show).join(' / ') : 'Unavailable'],
        [`${label(name)} realised nonzero mean (µS)`, stats && stats.zero_fraction < 1 ? show(stats.mean / (1 - stats.zero_fraction)) : 'Unavailable'],
        [`${label(name)} initial zero fraction / realised nonzero fraction`, [show(init?.initial_zero_fraction), density(name)].join(' / ')],
        [`${label(name)} realised mean / SD including zeros (µS)`, [stats?.mean, stats?.std].map(show).join(' / ')],
      ];
    }) },
    { label: 'Input and simulation', rows: [
      ['External input', 'One Poisson raster shared by Input→E and Input→I'],
      ['Fixed Input / recurrent connection probability', [cfg.input_connection_probability, cfg.connection_probability].map(show).join(' / ')],
      ['Constant rate per Input source (Hz)', show(cfg.external_rate_hz)],
      ['External seed offset', show(cfg.external_seed_offset)],
      ['Timestep / duration (ms)', [snapshot?.execution?.timestep_ms, cfg.duration_ms].map(show).join(' / ')],
      ['Execution seed', show(snapshot?.execution?.seed)],
      ['Execution device', show(snapshot?.execution?.device_resolved)],
      ['Requested compilation mode', cfg.performance?.compile_mode ?? 'Off'],
      ['CPU threads', show(cfg.torch_threads)],
      ['Spikes recorded', 'All E and I neurons; raw recording includes burn-in'],
      ['Excluded burn-in (ms)', show(cfg.burn_in_ms)],
      ['Measurement window (ms)', show(cfg.duration_ms - cfg.burn_in_ms)],
      ['Input-generation block length (steps)', show(cfg.recording_block_steps)],
      ['Training', snapshot?.execution?.parameter_membership?.trainable?.length === 0 ? 'None; gains follow prescribed native modulation' : 'Unavailable'],
    ] },
    { label: 'Control protocol', rows: [
      ['Scheduled parameter', label(cfg.control?.parameter ?? 'Unavailable')],
      ['Baseline / final gain', [cfg.control?.baseline_gain, cfg.control?.final_gain].map(show).join(' / ')],
      ['Absolute ramp start / end (ms)', [cfg.control?.ramp_start_ms, cfg.control?.ramp_end_ms].map(show).join(' / ')],
      ['Native gain sampling interval (ms)', show(cfg.control?.update_ms)],
      ['Baseline / ramp / hold (ms)', [cfg.baseline_ms, cfg.ramp_ms, cfg.hold_ms].map(show).join(' / ')],
    ] },
    { label: 'Analysis', rows: [
      ['Population-count bin width (ms)', show(cfg.analysis?.population_bin_ms)],
      ['Minimum peak separation (ms)', show(cfg.analysis?.peak_separation_ms)],
      ['Prominence fraction of maximum bin count', show(cfg.analysis?.peak_prominence_fraction)],
    ] },

  ];
})();

## C. Hold-window analysis

Pulse timing is measured independently in consecutive 100 ms intervals using
0.5 ms population-count bins, a minimum peak separation of 6 ms and prominence
of 12% of the largest count in each interval. This local threshold retains
small early volleys that the full-hold threshold can miss. Intervals crossing
100 ms boundaries are omitted. The last interval is shorter when the hold
duration is not a multiple of 100 ms. These descriptive measurements do not establish
a PING mechanism. Native recordings also retain population mean voltage and mean
conductance for each pathway. This runset contains no pathway-ablation controls.

## D. Mean-field parameters and numerical checks

The [mean-field note](/notes/mean-field-model) gives the model, gain,
[analytical gain slope](/notes/mean-field-model#derivation-gain-slope) and
[four-state reduction](/notes/mean-field-model#four-state-reductions).
The [main-text specialisation](#model-specialisation) defines exp148's pathways
and Input→I gain schedule. No exp117 numerical results are scientific inputs.

All six matrix sums use the referenced native parameter snapshot, including
zeros and excluded autapses; initializer means are not substituted. The gain
uses exp148's recorded capacitance, leak, reset/threshold voltages and refractory
periods. Its fixed noise scale is $\sigma_{V,Q}=4$ mV and rate-relaxation times
are $\tau_{r,E}=20$ ms and $\tau_{r,I}=5$ ms. For both target populations,
$\Delta V_{\mathrm{exc},Q}=70$ mV and $\Delta V_{\mathrm{inh},Q}=10$ mV.
These closure choices and couplings are not fitted to the observed transition.
Explicit transmission delays and weight heterogeneity are omitted; the note
describes the shared-input covariance and shunting approximations.

The note's equations use spikes/neuron/ms. Exp148's implementation and rate
figures use Hz: rates and gains are multiplied by 1,000, and rate inputs to
conductance filters are divided by 1,000. Solver tolerances below are stated
in the implementation's Hz units. The scheduled matrix gain $\alpha$ acts on
new external input; substituting its instantaneous stationary mean omits the
5 ms input-filtering lag during the ramp.

The current recipe uses SNNLab's native `NoisyLIF` gain with 48-point
Gauss–Legendre quadrature, symbolic Jacobian, bounded equilibrium continuation
and Brent stability-crossing refinement. Its matched-decay reduction retains
$g_e^I$ and $g_i^E$; reconstructed self-pathway conductances must satisfy the
native model's constraints. The recorded graph is copied before translating
only its Input→I modulation into an explicit numerical control. The original
spiking evidence remains unchanged.

The selected figure retains the numerical implementation recorded in its run.
Earlier figures used the experiment's local gain and Jacobian; those duplicate
implementations and comparison references have been removed. Protocol tests
now cover native model construction, modulation handling and exported units.
A complex crossing is an onset candidate; it does not determine criticality.

LSODA uses relative tolerance $10^{-8}$ and native absolute
rate tolerance $10^{-10}$ Hz. Conductance absolute tolerances multiply that rate
tolerance by $\tau_{E\to I}\sum_{j,k}\mathbf W^{E\to I}_{jk}/(1000N_I)$ or
$\tau_{I\to E}\sum_{j,k}\mathbf W^{I\to E}_{jk}/(1000N_E)$, respectively, in
µS/Hz. The maximum integration step is 0.5 ms.

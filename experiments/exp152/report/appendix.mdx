import { getEntry } from 'astro:content';
import Table from '../../../site/src/components/Table.astro';
import { imageUrl } from '../../../site/src/lib/pingstore';
import { reportSnapshots } from '../../../site/src/lib/report-data';

export const entry = await getEntry('reports', 'exp152');
export const native = entry?.data.runs?.exp152 ? await reportSnapshots('exp152', {
  analysisInput: 'analyse',
  snapshots: { controller: 'flexor/start/snapshot.json' },
  allowUnavailable: true,
}) : undefined;
export const graph = native?.snapshots.controller?.network;
export const cfg = native?.configuration;
export const show = value => value === undefined ? 'Unavailable' : typeof value === 'object' && value !== null && 'value' in value ? show(value.value) : String(value);
export const motor = graph?.populations?.find(row => row.id === 'motor');
export const slow = graph?.populations?.find(row => row.id === 'slow');
export const groups = [
  { label: 'Network', rows: [
    ['Motor / slow cells per controller', `${show(motor?.size)} / ${show(slow?.size)}`],
    ['Neuron model', show(cfg?.neuron_model)],


  ] },
  { label: 'Ionic membrane dynamics', rows: [
    ...[
      ['capacitance_nf', 'Motor / slow capacitance (nF)'],
      ['g_na_us', 'Motor / slow sodium conductance (µS)'],
      ['g_k_us', 'Motor / slow potassium conductance (µS)'],
      ['g_leak_us', 'Motor / slow leak conductance (µS)'],
      ['e_na_mv', 'Motor / slow sodium reversal (mV)'],
      ['e_k_mv', 'Motor / slow potassium reversal (mV)'],
      ['e_leak_mv', 'Motor / slow leak reversal (mV)'],
      ['excitatory_reversal_mv', 'Motor / slow excitation reversal (mV)'],
      ['inhibitory_reversal_mv', 'Motor / slow inhibition reversal (mV)'],
      ['initial_voltage_mv', 'Motor / slow initial voltage (mV)'],
    ].map(([name, label]) => [label, `${show(motor?.neuron?.config?.[name])} / ${show(slow?.neuron?.config?.[name])}`]),
    ['RK4 maximum internal step (ms)', show(cfg?.hh?.rk4_max_step_ms)],
  ] },
  { label: 'Synaptic decay (ms)', rows: (graph?.projections ?? []).map(projection => [projection.id, show(projection.synapse?.tau)]) },
  { label: 'Synapses and weights', rows: graph?.projections?.length ? graph.projections.map(projection => [projection.id, show(graph.parameters?.find(row => row.id === `${projection.id}.weight`)?.initializer?.value)]) : [['Snapshot evidence', 'Unavailable']] },
  { label: 'Input and simulation', rows: [
    ['Duration (s)', show(cfg?.duration)], ['Numerical timestep (ms)', show(cfg?.dt_ms)],
    ['Control period (ms)', show(cfg?.timer_ms)], ['Seed', show(cfg?.seed)],
    ['Starting pulse duration (ms)', show(cfg?.flexor_bias_ms)],
    ['Starting flexor / extensor drive (µS)', `${show(cfg?.flexor_initial_us)} / ${show(cfg?.extensor_initial_us)}`],
    ['Ongoing tonic drive (µS)', show(cfg?.tonic_us)],
    ['SNNLab execution', cfg?.compile_mode === null ? 'Eager' : show(cfg?.compile_mode)],
    ['Lag stop / delivery budget (ms)', `${show(cfg?.max_lag_ms)} / ${show(cfg?.max_spike_age_ms)}`],
  ] },
  { label: 'Joint mechanics', rows: Object.entries(cfg?.joint ?? { evidence: 'Unavailable' }).map(([name, value]) => [name, show(value)]) },
];
